package mod.moineau.contentpacks.util;

import mod.moineau.contentpacks.ContentPacks;
import net.fabricmc.loader.api.FabricLoader;
import net.minecraft.resources.Identifier;
import net.minecraft.resources.ResourceKey;
import net.minecraft.server.packs.resources.Resource;
import org.apache.commons.io.FileUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.helpers.MessageFormatter;

import java.io.File;
import java.io.IOException;
import java.util.LinkedList;
import java.util.List;
import java.util.stream.Collectors;

public final class ErrorLogger {
    public static final ErrorLogger LOAD = new ErrorLogger(ContentPacks.LOGGER, "errors", WarningType.SUMARY);
    public static final ErrorLogger OUTPUT = new ErrorLogger("ContentPacks/Output", "output_errors", WarningType.SUMARY);
    private final Logger logger;
    private final File file;
    private final List<String> lines = new LinkedList<>();
    private final WarningType type;

    public ErrorLogger(Logger logger, String fileName, WarningType type) {
        this.logger = logger;
        this.file = FabricLoader.getInstance().getGameDir().resolve("logs").resolve(ContentPacks.MOD_ID).resolve(fileName + ".log").toFile();
        this.type = type;
    }

    public ErrorLogger(String channel, String fileName, WarningType type) {
        this.logger = LoggerFactory.getLogger(channel);;
        this.file = FabricLoader.getInstance().getGameDir().resolve("logs").resolve(ContentPacks.MOD_ID).resolve(fileName + ".log").toFile();
        this.type = type;
    }

    public void write(String error) {
        this.lines.add(error);
        if (this.type == WarningType.ALL) this.logger.error(error);
    }

    public void write(String error, Object... args) {
        this.lines.add(MessageFormatter.basicArrayFormat(error, args));
    }

    public void write(ResourceKey<?> resourceKey, String error) {
        this.write("[{" + resourceKey.registry() + "} " + resourceKey.identifier() + "] " + error);
    }

    public void write(Identifier id, Resource resource, String error) {
        this.write("[(" + resource.sourcePackId() + ") " + id + "] " + error);
    }

    public void write(Identifier id, List<Resource> resources, String error) {
        this.write("[(" + resources.stream().map(Resource::sourcePackId).collect(Collectors.joining(" | ")) + ") " + id + "] " + error);
    }

    public void write() {
        if (this.type != WarningType.NONE && count() > 0) {
            this.logger.error("{} errors encountered!", this.count());
        }

        try {
            FileUtils.writeLines(this.file, this.lines);
        } catch (IOException e) {
            this.logger.error("Failed to write error file {}", file);
        }
    }

    public boolean hasErrors() {
        return !this.lines.isEmpty();
    }

    public int count() {
        return this.lines.size();
    }

    public enum WarningType {
        NONE,
        SUMARY,
        ALL
    }
}
